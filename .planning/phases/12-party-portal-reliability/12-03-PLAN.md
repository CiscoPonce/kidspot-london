     async searchEstablishments(params: {

     /**
      * Match venue to FHRS establishment
      */
     async matchFhrsToVenue(venue: {
       name: string;
       postcode?: string;
       lat?: number;
       lon?: number;
     }): Promise<FhrsMatchResult> {
       // Try exact postcode match first
       if (venue.postcode) {
         const postcodeResults = await this.searchEstablishments({
           postcode: venue.postcode,
           limit: 10,
         });

         // Look for name similarity
         for (const est of postcodeResults) {
           const similarity = this.calculateNameSimilarity(venue.name, est.BusinessName);
           if (similarity > 0.8) {
             return {
               establishment: est,
               confidence: 'high',
               match_type: 'fuzzy_name_postcode',
             };
           }
         }
       }

       // Try location-based search
       if (venue.lat && venue.lon) {
         const locationResults = await this.searchEstablishments({
           lat: venue.lat,
           lon: venue.lon,
           radius: 0.5, // 0.5 miles
           limit: 10,
         });

         for (const est of locationResults) {
           const similarity = this.calculateNameSimilarity(venue.name, est.BusinessName);
           if (similarity > 0.8) {
             return {
               establishment: est,
               confidence: 'medium',
               match_type: 'fuzzy_name_postcode',
             };
           }
         }
       }

       return {
         establishment: null,
         confidence: 'none',
         match_type: 'none',
       };
     },

     /**
      * Calculate name similarity using Levenshtein distance
      */
     calculateNameSimilarity(name1: string, name2: string): number {
       const normalize = (str: string) =>
         str.toLowerCase().replace(/[^a-z0-9]/g, '');

       const n1 = normalize(name1);
       const n2 = normalize(name2);

       if (n1 === n2) return 1.0;

       const maxLen = Math.max(n1.length, n2.length);
       if (maxLen === 0) return 1.0;

       // Simple Levenshtein distance
       const distance = this.levenshteinDistance(n1, n2);
       return 1 - distance / maxLen;
     },

     /**
      * Levenshtein distance implementation
      */
     levenshteinDistance(str1: string, str2: string): number {
       const m = str1.length;
       const n = str2.length;
       const dp = Array(m + 1).fill(null).map(() => Array(n + 1).fill(0));

       for (let i = 0; i <= m; i++) dp[i][0] = i;
       for (let j = 0; j <= n; j++) dp[0][j] = j;

       for (let i = 1; i <= m; i++) {
         for (let j = 1; j <= n; j++) {
           const cost = str1[i - 1] === str2[j - 1] ? 0 : 1;
           dp[i][j] = Math.min(
             dp[i - 1][j] + 1,
             dp[i][j - 1] + 1,
             dp[i - 1][j - 1] + cost
           );
         }
       }

       return dp[m][n];
     },

     /**
      * Check if FHRS business type is party-relevant
      */
     async isPartyRelevant(businessTypeId: number): Promise<boolean> {
       const result = await db.query(
         'SELECT is_party_relevant FROM fhrs_business_type_allowlist WHERE business_type_id = $1',
         [businessTypeId]
       );
       return result.rows[0]?.is_party_relevant || false;
     },
   };

   export async function matchVenueToFhrs(venueId: number): Promise<FhrsMatchResult> {
     // Get venue details
     const venueResult = await db.query(
       'SELECT id, name, postcode, lat, lon FROM venues WHERE id = $1',
       [venueId]
     );

     if (venueResult.rows.length === 0) {
       throw new Error('Venue not found');
     }

     const venue = venueResult.rows[0];

     // Match to FHRS
     const match = await fhrsService.matchFhrsToVenue({
       name: venue.name,
       postcode: venue.postcode,
       lat: venue.lat,
       lon: venue.lon,
     });

     // If match found, update venue
     if (match.establishment && match.confidence !== 'none') {
       await updateVenueFromFhrs(venueId, match.establishment, match.confidence);
     }

     return match;
   }

   async function updateVenueFromFhrs(
     venueId: number,
     establishment: FhrsEstablishment,
     confidence: 'high' | 'medium' | 'low'
   ): Promise<void> {
     // Check if editor_locked
     const isLocked = await checkEditorLocked(venueId);
     if (isLocked) {
       logger.info({ venueId, fhrsId: establishment.FHRSID }, 'Venue editor_locked - not updating from FHRS');
       return;
     }

     // Check if FHRS business type is party-relevant
     const isPartyRelevant = await fhrsService.isPartyRelevant(establishment.BusinessTypeID);
     if (!isPartyRelevant) {
       logger.info({ venueId, fhrsId: establishment.FHRSID, businessType: establishment.BusinessType }, 'FHRS business type not party-relevant - skipping');
       return;
     }

     // Get current venue data
     const currentVenue = await db.query('SELECT * FROM venues WHERE id = $1', [venueId]);
     const venue = currentVenue.rows[0];

     // Build address from FHRS data
     const addressParts = [
       establishment.AddressLine1,
       establishment.AddressLine2,
       establishment.AddressLine3,
       establishment.AddressLine4,
       establishment.PostCode,
     ].filter(Boolean);

     const fhrsAddress = addressParts.join(', ');

     // Update venue with FHRS data
     await db.query(
       `UPDATE venues
        SET fhrs_establishment_id = $1,
            address = COALESCE(NULLIF($2, ''), address),
            postcode = COALESCE(NULLIF($3, ''), postcode),
            updated_at = NOW()
        WHERE id = $4`,
       [establishment.FHRSID, fhrsAddress, establishment.PostCode, venueId]
     );

     await logProvenance({
       venue_id: venueId,
       field_name: 'fhrs_match',
       old_value: venue.fhrs_establishment_id?.toString() || null,

       reason: `Matched to FHRS establishment with ${confidence} confidence`,

     await storeFhrsEstablishment(establishment);
