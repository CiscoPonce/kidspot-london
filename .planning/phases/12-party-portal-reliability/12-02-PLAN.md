   CREATE OR REPLACE FUNCTION search_venues_by_facets(
     search_lat DOUBLE PRECISION,
     search_lon DOUBLE PRECISION,
     radius_meters DOUBLE PRECISION,
     facet_filter TEXT[] DEFAULT NULL,
     limit_count INTEGER DEFAULT 50
   )
   RETURNS TABLE (
     id BIGINT,
     source TEXT,
     source_id TEXT,
     name TEXT,
     type TEXT,
     lat DOUBLE PRECISION,
     lon DOUBLE PRECISION,
     distance_miles DOUBLE PRECISION,
     sponsor_tier TEXT,
     sponsor_priority INTEGER,
     kid_score NUMERIC,
     rating NUMERIC,
     price_level INTEGER,
     features JSONB,
     parent_facets TEXT[]
   ) AS $$
   BEGIN
     RETURN QUERY
     SELECT
       v.id,
       v.source,
       v.source_id,
       v.name,
       v.type,
       v.lat,
       v.lon,
       ST_Distance(ST_MakePoint(v.lon, v.lat)::geography, ST_MakePoint(search_lon, search_lat)::geography) / 1609.34 AS distance_miles,
       v.sponsor_tier,
       v.sponsor_priority,
       v.kid_score,
       v.rating,
       v.price_level,
       v.features,
       v.parent_facets
     FROM venues v
     WHERE v.is_active = TRUE
     AND ST_DWithin(ST_MakePoint(v.lon, v.lat)::geography, ST_MakePoint(search_lon, search_lat)::geography, radius_meters)
     AND (facet_filter IS NULL OR v.parent_facets && facet_filter)
     ORDER BY
       CASE
         WHEN v.sponsor_tier = 'gold' THEN 1
         WHEN v.sponsor_tier = 'silver' THEN 2
         WHEN v.sponsor_tier = 'bronze' THEN 3
         ELSE 4
       END,
       v.sponsor_priority DESC NULLS LAST,
       v.kid_score DESC NULLS LAST,
       ST_Distance(ST_MakePoint(v.lon, v.lat)::geography, ST_MakePoint(search_lon, search_lat)::geography) ASC
     LIMIT limit_count;
   END;
   $$ LANGUAGE plpgsql;
   ```

   export async function searchVenuesByFacets(
     query: FacetSearchQuery
   ): Promise<FacetSearchResponse> {
     const {
       lat,
       lon,
       radius_miles = 5,
       facets = [],
       limit = 50,
       borough,
       postcode
     } = query;

     // If no location provided, try to get from borough/postcode
     let searchLat = lat;
     let searchLon = lon;

     if ((!searchLat || !searchLon) && borough) {
       // Get centroid from borough (implement or use cached)
       const boroughCoords = await getBoroughCentroid(borough);
       if (boroughCoords) {
         searchLat = boroughCoords.lat;
         searchLon = boroughCoords.lon;
       }
     }

     if ((!searchLat || !searchLon) && postcode) {
       // Get coords from postcode using Postcodes.io
       const postcodeCoords = await getPostcodeCoords(postcode);
       if (postcodeCoords) {
         searchLat = postcodeCoords.lat;
         searchLon = postcodeCoords.lon;
       }
     }

     if (!searchLat || !searchLon) {
       return {
         success: false,
         data: { total: 0, sponsored: { count: 0, venues: [] }, regular: { count: 0, venues: [] }, all: [] },
         meta: {
           search: { lat: null, lon: null, radius_miles, radius_meters: radius_miles * 1609.34, facets: [], borough: borough || null },
           sponsor_info: { gold_count: 0, silver_count: 0, bronze_count: 0 },
           cache_hit: false
         }
       };
     }

     const radiusMeters = radius_miles * 1609.34;
     const facetArray = facets.length > 0 ? facets : null;

     const result = await db.query(
       'SELECT * FROM search_venues_by_facets($1, $2, $3, $4, $5)',
       [searchLat, searchLon, radiusMeters, facetArray, limit]
     );

     const venues: Venue[] = result.rows.map(row => ({
       id: row.id,
       name: row.name,
       lat: row.lat,
       lon: row.lon,
       type: row.type,
       source: row.source,
       source_id: row.source_id,
       slug: row.slug || `venue-${row.id}`,
       sponsor_tier: row.sponsor_tier,
       sponsor_priority: row.sponsor_priority,
       distance_miles: row.distance_miles,
       kid_score: row.kid_score,
       rating: row.rating,
       user_ratings_total: row.user_ratings_total,
       price_level: row.price_level,
       features: row.features,
       parent_facets: row.parent_facets,
     }));

     // Separate sponsored and regular
     const sponsored = venues.filter(v => v.sponsor_tier);
     const regular = venues.filter(v => !v.sponsor_tier);

     // Get sponsor stats
     const sponsorStats = await db.query('SELECT * FROM get_sponsor_stats()');
     const sponsorInfo = {
       gold_count: sponsorStats.rows.find(r => r.tier === 'gold')?.count || 0,
       silver_count: sponsorStats.rows.find(r => r.tier === 'silver')?.count || 0,
       bronze_count: sponsorStats.rows.find(r => r.tier === 'bronze')?.count || 0,
     };

     return {
       success: true,
       data: {
         total: venues.length,
         sponsored: {
           count: sponsored.length,
           venues: sponsored
         },
         regular: {
           count: regular.length,
           venues: regular
         },
         all: venues
       },
       meta: {
         search: {
           lat: searchLat,
           lon: searchLon,
           radius_miles,
           radius_meters: radiusMeters,
           facets: facets,
           borough: borough || null
         },
         sponsor_info: sponsorInfo,
         cache_hit: false
       }
     };
   }
   ```
